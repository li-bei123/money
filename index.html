📖 使用说明

  1. 收到招行短信 → 截图（可长截图多笔）→ 打开本页上传截图 → 自动识别 → 核对入账；也可以复制短信文字粘贴解析。
  2. 分类自动判断，预览时可以随手改；改过的商户会被记住。
  3. "明细"里点任何一笔可修改或删除；"图表"里按月看分类占比和每日走势。
  4. 手机浏览器菜单里选"添加到主屏幕"，就能像 App 一样从桌面打开。

云同步设置

和日历画报用同一个 Supabase 项目即可：项目网址、公钥填一样的；同步密钥点"随机生成"一个新的。第一次使用需先在 Supabase 里执行下面的建表语句（只需一次）。

第一步：建表语句（点开复制，到 Supabase 的 SQL Editor 执行） 
create table if not exists expense_ledger (
  board text primary key,
  data jsonb not null default '{}'::jsonb,
  updated_at bigint not null default 0
);
alter table expense_ledger enable row level security;
drop policy if exists "board access" on expense_ledger;
create policy "board access" on expense_ledger
  for all to anon
  using (board = (current_setting('request.headers', true)::json->>'x-board'))
  with check (board = (current_setting('request.headers', true)::json->>'x-board'));
